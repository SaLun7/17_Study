# 24. 실루엣 군집 평가

실루엣 계수와 다이어그램으로 군집 내부 응집도와 군집 간 분리도를 평가한다.

[← DAY7 목차](<README.md>) · [원본 노트북](<24_실루엣 군집 평가.ipynb>)

## 실습 내용

- 개별 샘플의 실루엣 계수를 이해한다.
- 군집별 평균과 전체 평균을 계산한다.
- 군집 수 2~5의 다이어그램을 비교한다.
- 내부 점수와 업무 해석을 함께 고려한다.

**데이터**  
Iris와 make_blobs 합성 데이터

## 핵심 메모

### a

같은 군집 샘플들과의 평균 거리다.

### b

가장 가까운 다른 군집까지의 평균 거리다.

### 실루엣 계수

(b-a)/max(a,b)이며 1에 가까울수록 응집·분리가 좋다.

```python
from sklearn.preprocessing import scale
from sklearn.datasets import load_iris
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_samples, silhouette_score
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

iris = load_iris()
feature_names = ['sepal_length','sepal_width','petal_length','petal_width']
df = pd.DataFrame(data=iris.data, columns=feature_names)
kmeans = KMeans(n_clusters=3, init='k-means++', max_iter=300,random_state=42).fit(df)

df['cluster'] = kmeans.labels_
 
score_samples = silhouette_samples(iris.data, df['cluster'])
print(score_samples.shape)

df['silhouette_coeff'] = score_samples
 
avg = silhouette_score(iris.data, df['cluster'])
print(avg)
```

```python
def visualize_silhouette(cluster_lists, X_features): 
    
    from sklearn.datasets import make_blobs
    from sklearn.cluster import KMeans
    from sklearn.metrics import silhouette_samples, silhouette_score

    import matplotlib.pyplot as plt
    import matplotlib.cm as cm
    import math
    
    # 입력값으로 클러스터링 갯수들을 리스트로 받아서, 각 갯수별로 클러스터링을 적용하고 실루엣 개수를 구함
    n_cols = len(cluster_lists)
    
    # plt.subplots()으로 리스트에 기재된 클러스터링 수만큼의 sub figures를 가지는 axs 생성 
    fig, axs = plt.subplots(figsize=(4*n_cols, 4), nrows=1, ncols=n_cols)
    
    # 리스트에 기재된 클러스터링 갯수들을 차례로 iteration 수행하면서 실루엣 개수 시각화
    for ind, n_cluster in enumerate(cluster_lists):
        
        # KMeans 클러스터링 수행하고, 실루엣 스코어와 개별 데이터의 실루엣 값 계산. 
        clusterer = KMeans(n_clusters = n_cluster, max_iter=500, random_state=42)
        cluster_labels = clusterer.fit_predict(X_features)
        
        sil_avg = silhouette_score(X_features, cluster_labels)
        sil_values = silhouette_samples(X_features, cluster_labels)
        
        y_lower = 10
        axs[ind].set_title('Number of Cluster : '+ str(n_cluster)+'\n' \
                          'Silhouette Score :' + str(round(sil_avg,3)) )
        axs[ind].set_xlabel("The silhouette coefficient values")
        axs[ind].set_ylabel("Cluster label")
        axs[ind].set_xlim([-0.1, 1])
# ... 이하 코드는 원본 노트북 참조
```
